WITH latest_reply AS (   -- MATERIALIZE: one recent, still-open reply per order BEFORE touching the 6.1M-row ORDERS table
  SELECT /*+ MATERIALIZE */ order_id, event_time, station_response_status_ids
    FROM ( SELECT sr.order_id, sr.event_time, sr.station_response_status_ids,
                  ROW_NUMBER() OVER (PARTITION BY sr.order_id ORDER BY sr.event_time DESC) AS rn
             FROM station_reply sr
            WHERE sr.event_time >= SYSDATE - 7 )     -- range-scans IDX_STATION_REPLY_EVENT_TIME
   WHERE rn = 1
     AND event_time <= SYSDATE - 8/24                -- latest reply must be at least 8h old
     AND INSTR(station_response_status_ids, lu.srs.PRJ_SORTER_DISCHARGE_CONFIRMED) = 0
)
SELECT * FROM (
SELECT /*+ LEADING(str ord) USE_NL(ord) */
       ord.order_nr, ord.order_id, ord.prj_alternate_lou_code, ord.loading_unit_code,

            THEN 'DISSAPPEARED AT CONSOLIDATION'

             AND SUBSTR(ord.order_nr, INSTR(ord.order_nr, '_') + 1) <> '1'

    AND lu.ops.to_token(ord.process_status_id) NOT IN ('CANCELLED')
    AND NOT EXISTS ( SELECT 1 FROM orders succ JOIN station_reply sr2 ON sr2.order_id = succ.order_id

                         = SUBSTR( ord.order_nr,  1, INSTR(ord.order_nr,  '_') - 1 )
                       AND succ.order_id <> ord.order_id AND succ.sheet_nr > 500
                       AND INSTR( sr2.station_response_status_ids, lu.srs.PRJ_SORTER_DISCHARGE_CONFIRMED ) > 0 )
ORDER BY str.event_time DESC )
WHERE categorization LIKE '%DISSAP%'
    OR categorization = 'SLAM KICKOUT PREMATURE RESHOP';
