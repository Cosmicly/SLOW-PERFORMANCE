/* =====================================================================================
   "Disappeared" / stuck order detection  --  rewrite
   -------------------------------------------------------------------------------------
   Changes vs. original:
     1. Dropped /*+ LEADING(str ord) USE_NL(ord) *"/ -- that hint was ~76M of the 78M cost.
     2. ROW_NUMBER() window  ->  aggregate + KEEP (DENSE_RANK LAST). Kills the
        WINDOW SORT PUSHED RANK and its 1437M of TempSpc.
     3. No SYSDATE-7 floor at all. The 8h rule is now a HAVING on MAX(event_time),
        so it is time-unbounded by design.
     4. lu.srs.* constants resolved ONCE in CTE k instead of ~15 function calls per row.
     5. lu.*.to_token() called once per surviving row, not ~10x inside the CASE.
     6. NOT EXISTS rewritten from SUBSTR(a)=SUBSTR(b) (unindexable -> two full scans,
        VW_SQ_1 cost 852K) to a prefix LIKE, which range-scans IDX_TRO_ORDER_NR_SHEET_NR.
     7. NOT EXISTS moved AFTER categorization, so it runs on ~1.7K rows not 46M.
     8. PRJ_STATION_PRIORITY LISTAGG computed once instead of twice.
     9. 'GOOD UDC + 32R' branch removed -- unreachable, confirmed rows are filtered in k.

   READ BEFORE RUNNING -- two behaviour notes flagged inline below:
     - the CONSOLIDATION branch had literal '%' inside an IN list (a no-op wildcard)
     - INSTR on a concatenated id list can substring-match (id 5 matches id 15)
   ===================================================================================== */

WITH k AS (                       /* status ids evaluated once, not once per row per INSTR */
  SELECT /*+ MATERIALIZE */
         TO_CHAR(lu.srs.PRJ_SORTER_DISCHARGE_CONFIRMED) AS confirmed,
         TO_CHAR(lu.srs.PRJ_SORTER_DISCHARGE_FAILED)    AS failed,
         TO_CHAR(lu.srs.PRJ_SORTER_DISCHARGE_TO_ERROR)  AS to_error,
         TO_CHAR(lu.srs.PRJ_SORTER_DISCHARGE_ATTEMPT)   AS attempt
    FROM dual
),

last_reply AS (                   /* one row per order = its most recent reply.
                                     No day cap. Only rule: that reply is >= 8h old
                                     and is not a discharge-confirmed. */
  SELECT /*+ MATERIALIZE NO_MERGE */
         lr.order_id, lr.event_time, lr.ids
    FROM ( SELECT sr.order_id,
                  MAX(sr.event_time) AS event_time,
                  MAX(sr.station_response_status_ids)
                      KEEP (DENSE_RANK LAST ORDER BY sr.event_time) AS ids
             FROM station_reply sr
            /* optional floor if you ever want one:  WHERE sr.event_time >= SYSDATE - :days */
            GROUP BY sr.order_id
           HAVING MAX(sr.event_time) <= SYSDATE - 8/24 ) lr
   CROSS JOIN k
   WHERE INSTR(lr.ids, k.confirmed) = 0
),

cand AS (                         /* join to ORDERS -- let the CBO pick HASH here.
                                     to_token() now evaluated once per row, up front. */
  SELECT ord.order_nr, ord.order_id, ord.prj_alternate_lou_code, ord.loading_unit_code,
         ord.sheet_nr, ord.delivery_date, ord.last_read_station_id,
         lu.lut.to_token(ord.loading_unit_type_id) AS load_carrier,
         lu.coi.to_token(ord.last_control_info_id) AS control_info,
         lu.sta.to_token(ord.last_read_station_id) AS station_divert,
         CASE WHEN INSTR(ord.order_nr, '_') > 0
              THEN SUBSTR(ord.order_nr, 1, INSTR(ord.order_nr, '_') - 1) END AS base_order_nr,
         CASE WHEN INSTR(ord.order_nr, '_') > 0
              THEN SUBSTR(ord.order_nr, INSTR(ord.order_nr, '_') + 1) END AS order_suffix,
         str.event_time, str.ids,
         k.confirmed, k.failed, k.to_error, k.attempt
    FROM last_reply str
    JOIN orders ord ON ord.order_id = str.order_id
   CROSS JOIN k
   WHERE ord.process_status_id >= 50
     AND lu.ops.to_token(ord.process_status_id)      <> 'CANCELLED'
     AND lu.lut.to_token(ord.loading_unit_type_id) NOT IN ('FULL', 'RECCARTON')
),

cat AS (                          /* categorise. Branch order preserved exactly.
                                     'confirmed = 0' terms dropped -- already guaranteed. */
  SELECT c.*,
         CASE WHEN INSTR(c.ids, c.failed)   > 0 THEN 'PRJ_SORTER_DISCHARGE_FAILED'
              WHEN INSTR(c.ids, c.to_error) > 0 THEN 'PRJ_SORTER_DISCHARGE_TO_ERROR'
              WHEN INSTR(c.ids, c.attempt)  > 0 THEN 'PRJ_SORTER_DISCHARGE_ATTEMPT'
         END AS last_sorter_status,
         CASE
           WHEN c.station_divert LIKE '%BIAR%'
                THEN 'DISSAPPEARED NO MESSAGE GETTING ON SORTER'

           WHEN INSTR(c.ids, c.attempt)  > 0
            AND INSTR(c.ids, c.failed)   = 0
            AND INSTR(c.ids, c.to_error) = 0
                THEN 'DISSAPPEARED NO BEUMER UDC'

           WHEN INSTR(c.ids, c.failed) > 0
                THEN 'DISSAPPEARED AFTER FAILED DIVERT'

           WHEN NVL(INSTR(c.ids, c.failed),   0) = 0
            AND NVL(INSTR(c.ids, c.to_error), 0) = 0
            AND NVL(INSTR(c.ids, c.attempt),  0) = 0
            AND c.load_carrier IN ('SHIPPINGCARTON', 'BAG')
            AND c.sheet_nr > 500
                THEN 'DISSAPPEARED AT JACKPOT'

           /* ---- FLAG: original was IN ('%CONS01F2LAAP001%','CONS01F2ERRO001').
                   The % chars are literal inside IN, so that first value never matched.
                   Rewritten as a real LIKE. Revert to '=' if the old behaviour was intended. */
           WHEN c.station_divert LIKE '%CONS01F2LAAP001%'
             OR c.station_divert = 'CONS01F2ERRO001'
                THEN 'DISSAPPEARED AT CONSOLIDATION'

           WHEN c.station_divert LIKE 'SLA%READ%'
            AND c.order_suffix IS NOT NULL
            AND c.order_suffix <> '1'
            AND EXISTS ( SELECT 1
                           FROM orders o1
                           JOIN station_reply s1 ON s1.order_id = o1.order_id
                          WHERE o1.order_nr  = c.base_order_nr || '_1'
                            AND o1.sheet_nr <= 500
                            AND o1.order_id <> c.order_id
                            AND INSTR(s1.station_response_status_ids, c.confirmed) > 0 )
                THEN 'SLAM KICKOUT PREMATURE RESHOP'

           WHEN c.station_divert LIKE 'SLA%READ%'  THEN 'DISSAPPEARED AT SLAM'
           WHEN c.station_divert LIKE '%BAGG%'     THEN 'DISSAPPEARED AT BAGGING'
           WHEN c.station_divert IN ('OUTB01F1SCAN002', 'OUTB01F1SCAN001')
                                                   THEN 'DISSAPPEARED ON RECIRC'
           WHEN c.station_divert LIKE 'IFM%EXIT%'  THEN 'BEFORE SLAM DISSAPPEARED AT CACL EXIT'
           WHEN c.station_divert LIKE 'OUT%BEIF%'  THEN 'DISSAPPEARED AT SORTER INFEED'
         END AS categorization
    FROM cand c
),

final AS (                        /* anti-join runs LAST, on the categorised survivors only */
  SELECT c.order_nr, c.order_id, c.prj_alternate_lou_code, c.loading_unit_code,
         c.sheet_nr, c.delivery_date, c.load_carrier, c.control_info,
         c.station_divert, c.event_time, c.last_sorter_status, c.categorization,
         ( SELECT LISTAGG(lu.sta.to_token(pri.station_id), ', ')
                    WITHIN GROUP (ORDER BY pri.target_station_priority)
             FROM prj_station_priority pri
            WHERE pri.order_id = c.order_id ) AS target_station_prios
    FROM cat c
   WHERE c.categorization IS NOT NULL
     AND NOT EXISTS ( SELECT 1                       /* successor sheet already shipped? */
                        FROM orders succ
                        JOIN station_reply sr2 ON sr2.order_id = succ.order_id
                       WHERE succ.order_nr LIKE c.base_order_nr || '\_%' ESCAPE '\'
                         AND succ.sheet_nr  > 500
                         AND succ.order_id <> c.order_id
                         AND INSTR(sr2.station_response_status_ids, c.confirmed) > 0 )
)

SELECT f.order_nr, f.order_id, f.prj_alternate_lou_code, f.loading_unit_code,
       f.sheet_nr, f.delivery_date, f.load_carrier, f.control_info,
       f.station_divert,
       TO_CHAR(f.event_time, 'DD-MON-YY HH.MI.SSXFF PM') AS event_time,
       f.target_station_prios,
       CASE WHEN INSTR(f.target_station_prios, f.station_divert) > 0
            THEN 'Y' ELSE 'N' END AS divert_matches_prio,
       f.last_sorter_status,
       f.categorization
  FROM final f
 ORDER BY f.event_time DESC;


/* =====================================================================================
   SUPPORTING NOTES
   -------------------------------------------------------------------------------------
   A. If STATION_RESPONSE_STATUS_IDS is a CLOB, MAX(...) KEEP (DENSE_RANK LAST ...) will
      raise ORA-00932. Fallback -- keep ROW_NUMBER but only over candidate orders:

        SELECT order_id, event_time, ids FROM (
          SELECT sr.order_id, sr.event_time, sr.station_response_status_ids AS ids,
                 ROW_NUMBER() OVER (PARTITION BY sr.order_id ORDER BY sr.event_time DESC) rn,
                 MAX(sr.event_time) OVER (PARTITION BY sr.order_id) AS last_evt
            FROM station_reply sr )
         WHERE rn = 1 AND last_evt <= SYSDATE - 8/24;

   B. Tie-break difference: KEEP DENSE_RANK LAST takes MAX(ids) among rows sharing the
      newest event_time; ROW_NUMBER took an arbitrary one. Only matters on exact ties.

   C. The plan claimed IDX_STATION_REPLY_EVENT_TIME but did TABLE ACCESS FULL and
      estimated 46M rows surviving event_time >= SYSDATE-7. Check:
        SELECT index_name, status, visibility FROM user_indexes
         WHERE table_name = 'STATION_REPLY';
        SELECT num_rows, last_analyzed FROM user_tables
         WHERE table_name IN ('STATION_REPLY','ORDERS');
      Stale stats there are why the CBO thought a 46M-row NL was acceptable.

   D. Covering index to turn the group-by scan into an INDEX FAST FULL SCAN:
        CREATE INDEX idx_sr_order_evt ON station_reply (order_id, event_time)
          COMPRESS 1 ONLINE;
      Won't avoid reading the table for the status ids, but makes MAX(event_time)
      per order much cheaper if you ever split it into two passes.

   E. If lu.ops / lu.lut expose id constants the way lu.srs does (e.g. lu.ops.CANCELLED),
      swap the to_token() predicates for numeric comparisons:
        AND ord.process_status_id     <> lu.ops.CANCELLED
        AND ord.loading_unit_type_id NOT IN (lu.lut.FULL, lu.lut.RECCARTON)
      That removes a PL/SQL context switch per row and makes the columns indexable.

   F. If the to_token functions aren't declared DETERMINISTIC, declaring them so lets
      Oracle cache repeated calls -- station/type/status ids are a small closed set.

   G. INSTR on a delimited id list substring-matches: id 5 matches inside id 15 or 51.
      If the column is comma-delimited, the safe form is:
        INSTR(',' || c.ids || ',', ',' || c.confirmed || ',') > 0
      Worth checking against real values before you trust the counts.
   ===================================================================================== */
